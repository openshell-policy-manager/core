import json
import sys

config_path = sys.argv[1]
with open(config_path) as f:
    config = json.load(f)

skills = config.get('agent_skills', {})
print(f"  {len(skills)} Skills konfiguriert:")
for skill, data in skills.items():
    enabled = "✓" if data.get('enabled', False) else "✗"
    perms = ", ".join(data.get('permissions', []))
    print(f"    $enabled {skill}: [{perms}]")
PYEOF
